💻 . Generación y Validación de JWT con `python-jose`: Código en Vivo

💡 Concepto Clave:

"El flujo JWT es como una discoteca: verificas tu DNI en la entrada (login) y te dan una pulsera (JWT) para pedir consumiciones en..."

📌 Explicación Técnica:

`python-jose` es la librería estándar para firmar y verificar tokens en el ecosistema de FastAPI.

sec-9.py
Python 
from datetime import datetime, timedelta
from jose import jwt, JWTError

SECRET_KEY = "mi_clave_secreta_super_segura_para_desarrollo"
ALGORITHM = "HS256"

def create_access_token(data: dict, expires_delta: timedelta = None) -> str:
    to_encode = data.copy()
    expire = datetime.utcnow() + (expires_delta or timedelta(minutes=15))
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

def verify_access_token(token: str) -> dict:
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        return payload
    except JWTError:
        # Captura firmas inválidas, tokens expirados o malformados
        return None

from tinydb import TinyDB, Query

db = TinyDB("db.json")
users_table = db.table("users")
profiles_table = db.table("profiles")

# Crear un usuario y su perfil vinculado
def register_user_with_profile(email: str, hashed_pass: str, name: str):
    user_id = "user_uuid_generado_123" # Usa uuid.uuid4() en producción
    
    # Insertar en tabla de credenciales
    users_table.insert({
        "id": user_id,
        "email": email,
        "hashed_password": hashed_pass
    })
    
    # Insertar en tabla de perfiles (Relación 1-a-1 mediante user_id)
    profiles_table.insert({
        "id": "profile_uuid_456",
        "user_id": user_id,
        "name": name
    })
